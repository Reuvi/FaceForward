cmake_minimum_required(VERSION 3.16)
if(APPLE AND NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "10.15" CACHE STRING "Minimum macOS version")
endif()
project(face_input_driver VERSION 0.1.0 LANGUAGES C CXX)
include(CTest)

find_package(Threads REQUIRED)
option(FID_NATIVE_BACKEND "Build the host OS input backend" ON)

add_library(face_input_driver SHARED src/driver.cpp src/control.cpp)
target_compile_features(face_input_driver PUBLIC cxx_std_17)
target_include_directories(face_input_driver PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_include_directories(face_input_driver PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_link_libraries(face_input_driver PRIVATE Threads::Threads)
set_target_properties(face_input_driver PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)

if(FID_NATIVE_BACKEND AND APPLE)
    target_sources(face_input_driver PRIVATE src/backend_macos.cpp)
    target_link_libraries(face_input_driver PRIVATE "-framework ApplicationServices" "-framework CoreGraphics")
    target_compile_definitions(face_input_driver PRIVATE FID_HAS_NATIVE=1)
elseif(FID_NATIVE_BACKEND AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(LIBEVDEV REQUIRED IMPORTED_TARGET libevdev)
    pkg_check_modules(X11 QUIET IMPORTED_TARGET x11)
    pkg_check_modules(XRANDR QUIET IMPORTED_TARGET xrandr)
    target_sources(face_input_driver PRIVATE src/backend_linux.cpp)
    target_link_libraries(face_input_driver PRIVATE PkgConfig::LIBEVDEV)
    if(X11_FOUND)
        target_link_libraries(face_input_driver PRIVATE PkgConfig::X11)
        target_compile_definitions(face_input_driver PRIVATE FID_HAS_X11=1)
        if(XRANDR_FOUND)
            target_link_libraries(face_input_driver PRIVATE PkgConfig::XRANDR)
            target_compile_definitions(face_input_driver PRIVATE FID_HAS_XRANDR=1)
        endif()
    else()
        message(STATUS "X11 development files not found; Linux target mode unavailable")
    endif()
    target_compile_definitions(face_input_driver PRIVATE FID_HAS_NATIVE=1)
else()
    message(STATUS "Building dry-run backend only")
endif()

if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(face_input_driver PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(driver_demo examples/demo.cpp)
target_link_libraries(driver_demo PRIVATE face_input_driver)

if(BUILD_TESTING)
    add_executable(driver_tests tests/control_tests.cpp src/control.cpp)
    target_compile_features(driver_tests PRIVATE cxx_std_17)
    target_include_directories(driver_tests PRIVATE include src)
    add_test(NAME control COMMAND driver_tests)
    add_executable(driver_api_tests tests/api_tests.cpp)
    target_link_libraries(driver_api_tests PRIVATE face_input_driver)
    add_test(NAME c_api COMMAND driver_api_tests)
    add_executable(driver_c_header_test tests/c_header_test.c)
    target_link_libraries(driver_c_header_test PRIVATE face_input_driver)
    add_test(NAME c_header COMMAND driver_c_header_test)
    find_package(Python3 3.9 COMPONENTS Interpreter QUIET)
    if(Python3_Interpreter_FOUND)
        add_test(NAME testing_console
            COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_testing_console.py
                    --library $<TARGET_FILE:face_input_driver>)
        set_tests_properties(testing_console PROPERTIES TIMEOUT 45)
        add_test(NAME processing_adapter
            COMMAND ${Python3_EXECUTABLE} -m unittest discover
                    -s ${CMAKE_CURRENT_SOURCE_DIR}/tests -p test_output_adapter.py)
        set_tests_properties(processing_adapter PROPERTIES TIMEOUT 15
            WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/..
            ENVIRONMENT "FID_LIBRARY=$<TARGET_FILE:face_input_driver>")
    else()
        message(STATUS "Python 3.9+ not found; skipping terminal tester checks")
    endif()
endif()
